    @{ Name = 'known completed negative run'; Value = @{ noFurtherCallsPending = $true; requiresReconciliation = $false; error = 'candidate rejected' }; Pass = $true },
    @{ Name = 'known completed successful run'; Value = @{ noFurtherCallsPending = $true; requiresReconciliation = $false }; Pass = $true },
    @{ Name = 'local terminal but unknown native effects'; Value = @{ noFurtherCallsPending = $true; requiresReconciliation = $true }; Pass = $false },
    @{ Name = 'active local calls'; Value = @{ noFurtherCallsPending = $false; requiresReconciliation = $false }; Pass = $false },
    @{ Name = 'missing native reconciliation state'; Value = @{ noFurtherCallsPending = $true }; Pass = $false },
    @{ Name = 'missing local completion state'; Value = @{ requiresReconciliation = $false }; Pass = $false },
    @{ Name = 'string booleans refused'; Value = @{ noFurtherCallsPending = 'true'; requiresReconciliation = 'false' }; Pass = $false }
)

}
