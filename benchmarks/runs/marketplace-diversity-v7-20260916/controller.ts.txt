import { runDiversityBenchmark, type DiversityCase } from "../scripts/run-marketplace-diversity-benchmark";
const [caseId,label,priorCharged,priorReservations]=process.argv.slice(2);
const planners={gemini:"google/gemini-3.7-flash",sol:"openai/gpt-5.6-sol"} as const;
if(!Object.hasOwn(planners,label)) throw Error("Expected explicit registered planner label");
const result=await runDiversityBenchmark(caseId as DiversityCase,{
  version:14,mode:"plan",plannerModel:planners[label as keyof typeof planners],
  model:"google/gemini-3.7-flash",
  studioId:"af74e97f-57b3-40c6-a281-83e1040ff755",
  output:`benchmarks/runs/marketplace-diversity-v7-20260916/${caseId}/${label}`,
  priorChargedMicros:Number(priorCharged),priorReservationsMicros:Number(priorReservations),
  budgetMicros:1_000_000,baseUrl:"http://127.0.0.1:4335",dataDirectory:".forge/diversity-v7-runtime",
});
console.log(JSON.stringify(result,null,2));
if(!result.noFurtherCallsPending||!result.routesRestored||result.requiresReconciliation) process.exitCode=2;
else if(result.error) process.exitCode=1;
