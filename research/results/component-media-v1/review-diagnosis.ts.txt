import fs from "node:fs";
import path from "node:path";
import {createHash} from "node:crypto";
import {loadComponentReviewEvidence, validateComponentReview} from "../src/generation/component-review";
const base="benchmarks/runs/marketplace-diversity-v8-20260916/bubble-wrap";
const project=JSON.parse(fs.readFileSync(path.join(base,"final-project.json"),"utf8"));
const reviews=fs.readFileSync(path.join(base,"model-events.jsonl"),"utf8").trim().split("\n").map((s)=>JSON.parse(s)).filter((r)=>r.phase==="reviewer");
const results=reviews.map((r,i)=>{
 const responseSha256=createHash("sha256").update(r.response).digest("hex");
 let parsed:any;
 try {parsed=JSON.parse(r.response);} catch(error){return {attempt:i+1,responseSha256,jsonValid:false,error:String(error)};}
 const evidence=loadComponentReviewEvidence(path.join(base,"asset-evidence"),parsed.packetHash,parsed.inputHash);
 const need=project.assetPipeline.needs.find((n:any)=>n.kind==="Model");
 let validationError:string|undefined;
 try{validateComponentReview(parsed,evidence,[...new Set([need.requirementId,...need.intent.relatedRequirementIds])] as string[]);}catch(error){validationError=String(error);}
 const mismatches=parsed.sources.flatMap((source:any)=>{
   const body=evidence.sourceBodies.find(b=>b.sha256===source.sha256);
   return source.dependencies.filter((d:any)=>!body?.source.includes(d.sourceQuote)).map((d:any)=>({
     bodySha256:source.sha256,kind:d.kind,value:d.value,quote:d.sourceQuote,
     exactMatch:false,crlfOnlyMismatch:body?.source.replaceAll("\r\n","\n").includes(d.sourceQuote.replaceAll("\r\n","\n")),
     whitespaceOnlyMatch:body?.source.replace(/\s+/g," ").includes(d.sourceQuote.replace(/\s+/g," ")),
   }));
 });
 return {attempt:i+1,responseSha256,jsonValid:true,validationError,disposition:parsed.disposition,mismatches};
});
fs.writeFileSync(path.join(base,"review-diagnosis.json"),JSON.stringify({readOnly:true,responsesChanged:false,results},null,2)+"\n");
console.log(JSON.stringify(results,null,2));
