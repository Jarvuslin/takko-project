import fs from "node:fs";
import path from "node:path";
import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { componentReviewModelEvidence, validateComponentReview, extractComponentSourceLines } from "../src/generation/component-review";

const read = (file: string) => JSON.parse(fs.readFileSync(file, "utf8"));
const hash = (value: string) => createHash("sha256").update(value).digest("hex");
const root = "research/results";
const original = read(path.join(root, "source-review-sol-v1/offline/input.json"));
const outcomes: unknown[] = [];
for (const name of ["source-review-sol-v1", "source-review-lines-v1"]) {
  const directory = path.join(root, name, "live");
  const protocol = read(path.join(directory, "protocol.json"));
  const input = read(path.join(directory, "input.json"));
  const result = read(path.join(directory, "result.json"));
  const calls = read(path.join(directory, "calls.json"));
  const before = read(path.join(directory, "key-before.json"));
  const after = read(path.join(directory, "key-after-settled.json"));
  assert.equal(input.request, original.request);
  assert.deepEqual(input.gameContext, original.gameContext);
  assert.deepEqual(input.context.need, original.context.need);
  assert.deepEqual(input.context.requirementIds, original.context.requirementIds);
  assert.deepEqual(input.context.evidence, name === "source-review-lines-v1" ? componentReviewModelEvidence(original.context.evidence) : original.context.evidence);
  assert.equal(protocol.nativeOperations, 0);
  assert.equal(result.nativeGameplay, false);
  assert.equal(result.rawGamePass, false);
  assert.equal(result.unknownLiability, false);
  assert.equal(result.inFlightReservations, 0);
  assert(calls.length <= 2);
  const reviews = [];
  let cost = 0;
  for (const call of calls) {
    const request = read(path.join(directory, `call-${call.id}-request.json`));
    const wireText = fs.readFileSync(path.join(directory, `call-${call.id}-response.txt`), "utf8");
    assert(!/sk-or-v1-[a-f0-9]{30,}/i.test(wireText + JSON.stringify(request)));
    const wire = JSON.parse(wireText);
    assert.equal(request.model, protocol.profile.model);
    assert.equal(request.max_tokens, 12000);
    assert.equal(call.chargedMicros, Math.ceil(wire.usage.cost * 1e6));
    assert.equal(call.status, 200);
    cost += wire.usage.cost;
    const raw = wire.choices[0].message.content;
    let decision: any, error: string | undefined;
    try {
      decision = validateComponentReview(JSON.parse(raw), original.context.evidence, original.context.requirementIds);
    } catch (e) { error = (e as Error).message; }
    const dependencies = decision?.sources.flatMap((s: any) => s.dependencies.map((d: any) => ({ sourceHash: s.sha256, ...d }))) ?? [];
    const citations = dependencies.filter((d: any) => d.sourceLines).map((d: any) => ({
      sourceHash: d.sourceHash, kind: d.kind, value: d.value, range: d.sourceLines,
      exactText: extractComponentSourceLines(original.context.evidence.sourceBodies.find((s: any) => s.sha256 === d.sourceHash).source, d.sourceLines),
    }));
    if (name === "source-review-sol-v1") {
      assert.match(error!, call.id === 1 ? /every removed capability exactly once/ : /exact source quotation/);
      const rawReview = JSON.parse(raw);
      assert(rawReview.sources.some((s: any) => s.dependencies.some((d: any) => d.sourceQuote && !original.context.evidence.sourceBodies.find((b: any) => b.sha256 === s.sha256).source.includes(d.sourceQuote))), "Both Sol responses contain the observed splice, although permission coverage fails first on attempt 1");
    }
    if (call.id === calls.length) {
      assert.equal(!!decision, result.contractPassed);
      if (decision) assert.deepEqual(read(path.join(directory, "decision.json")), decision);
    }
    reviews.push({ attempt: call.id, responseSha256: hash(raw), contractPassed: !!decision, error, lineCitations: citations, legacyQuoteCount: dependencies.length - citations.length });
  }
  assert(Math.abs(after.usage - before.usage - cost) < 0.000002, "Official key delta must settle to known provider receipts");
  const charged = calls.reduce((n: number, c: any) => n + c.chargedMicros, 0);
  assert.equal(result.chargedOrUnknownMicros, charged);
  assert.equal(result.conservativePriorMicros, Math.max(protocol.priorMicros, Math.ceil(before.usage * 1e6)) + charged);
  outcomes.push({ name, providerCost: cost, officialDelta: after.usage - before.usage, remaining: after.remaining, reviews });
}
const destination = path.join(root, "source-review-lines-v1/verification.json");
assert(!fs.existsSync(destination), "Preserve previous verification");
fs.writeFileSync(destination, JSON.stringify({ checkedAt: new Date().toISOString(), nativeOperations: 0, rawOutputsChanged: false, outcomes }, null, 2) + "\n");
console.log(JSON.stringify(outcomes.map(({name, providerCost, remaining, reviews}: any) => ({name, providerCost, remaining, reviews: reviews.map((r: any) => ({attempt:r.attempt, contractPassed:r.contractPassed, error:r.error,lineCitations:r.lineCitations.length,legacyQuoteCount:r.legacyQuoteCount}))})), null, 2));
