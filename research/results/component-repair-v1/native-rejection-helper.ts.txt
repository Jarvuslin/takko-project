import assert from "node:assert/strict";
export async function expectNativeRejection(operation:()=>Promise<unknown>,markUncertain:()=>void,pattern?:RegExp){
 let rejected:any;
 try{await operation();}catch(error){
  rejected=error;
  if((error as any)?.effects==="unknown"){markUncertain();throw error;}
 }
 assert(rejected,"Expected operation to reject");
 if(pattern)assert.match(String(rejected),pattern);
 return String(rejected);
}
