import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { reconstructSecondAdaptation } from './component-repair-route-probe';

const [mode, afterFile, outputFile = 'verification.json'] = process.argv.slice(2);
assert(mode === 'offline' || mode === 'live');
assert.equal(path.basename(outputFile), outputFile);
const root = 'research/results/component-repair-routing-v1';
const dir = path.join(root, mode);
const read = (name: string) => JSON.parse(fs.readFileSync(path.join(dir, name), 'utf8'));
const hash = (bytes: string | Buffer) => createHash('sha256').update(bytes).digest('hex');
const result = read('result.json'), protocol = read('protocol.json'), reservation = read('reservation.json');
assert.equal(result.chargedMicros, mode === 'live' ? reservation.chargedMicros : 0);
assert.equal(result.liabilityMicros, mode === 'live' ? reservation.liabilityMicros : 0);
assert.equal(result.generationId, reservation.generationId ?? null);
assert.equal(result.mode, mode);
assert.equal(result.localRequestSettled, true);
assert.equal(result.noFurtherRequests, true);
assert.equal(result.nativeOperations, 0);
assert.equal(result.rawGamePass, false);
assert.equal(result.calls, 1);
assert.equal(result.actualPaidCalls, mode === 'live' ? 1 : 0);
if (mode === 'live') assert.equal(fs.existsSync('.forge/component-repair-route-probe.lock'), false);
assert.equal(hash(fs.readFileSync('.forge/component-repair-route-probe.ts')), protocol.controllerSha256);
for (const [file, digest] of Object.entries(protocol.sources)) assert.equal(hash(fs.readFileSync(file)), digest);
const body = fs.readFileSync(path.join(dir, 'request-body.json'), 'utf8');
assert.equal(hash(body), protocol.requestSha256);
const historical = read('historical-gemini-request-body.json');
historical.model = protocol.profile.model;
assert.equal(JSON.stringify(historical), body);
const reconstruction = reconstructSecondAdaptation(path.join(dir, 'asset-evidence'));
assert.equal(hash(JSON.stringify(reconstruction.context)), protocol.contextSha256);
assert.deepEqual(JSON.parse(JSON.parse(body).messages[1].content), JSON.parse(JSON.stringify(reconstruction.context)));
let decisionIndependentlyValid = false;
if (fs.existsSync(path.join(dir, 'decision.json'))) {
  const validated = reconstruction.validate(fs.readFileSync(path.join(dir, 'raw-answer.txt'), 'utf8'));
  assert.deepEqual(JSON.parse(JSON.stringify(validated)), read('decision.json'));
  decisionIndependentlyValid = true;
}
if (result.contractPassed) {
  assert(decisionIndependentlyValid);
  assert.equal(reservation.status, 'validated');
  assert.equal(reservation.liabilityMicros, reservation.chargedMicros);
} else {
  assert(reservation.liabilityMicros >= reservation.reserveMicros);
}
assert.equal(result.providerOutcomeUnknown, mode === 'live' && reservation.dispatched && reservation.chargedMicros === null);
assert.equal(result.stopDoesNotProveRemoteSettlement, result.providerOutcomeUnknown);
assert.equal(result.conservativePriorMicros, result.priorMicros + result.liabilityMicros);
let officialDeltaUsd: number | null = null;
if (mode === 'live') {
  assert(afterFile, 'A fresh redacted official key snapshot is required');
  assert.equal(body, fs.readFileSync(path.join(root, 'offline/request-body.json'), 'utf8'));
  assert.deepEqual(read('reconstruction-proof.json'), JSON.parse(fs.readFileSync(path.join(root, 'offline/reconstruction-proof.json'), 'utf8')));
  const before = read('key-before.json'), after = read(afterFile);
  officialDeltaUsd = after.usage - before.usage;
  assert(officialDeltaUsd >= -1e-9);
  if (reservation.chargedMicros !== null) {
    const response = read('response.txt');
    assert.equal(Math.ceil(response.usage.cost * 1e6), reservation.chargedMicros);
    assert.equal(response.id ?? null, reservation.generationId);
    assert(Math.abs(officialDeltaUsd - response.usage.cost) <= 0.000001, 'Aggregate change differs from known receipt; reconcile without erasing liabilities');
  } else assert(officialDeltaUsd * 1e6 <= reservation.liabilityMicros + 1);
}
let files = 0;
function inspect(directory: string) {
  for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
    const file = path.join(directory, entry.name);
    if (entry.isDirectory()) inspect(file);
    else { files++; assert(!/sk-or-v1-[a-f0-9]{64}/i.test(fs.readFileSync(file).toString('utf8')), 'Plaintext key detected; value suppressed'); }
  }
}
inspect(dir);
const verification = { at: new Date().toISOString(), mode, files, sourcePins: Object.keys(protocol.sources).length, requestAndContextVerified: true, contractPassed: result.contractPassed, decisionIndependentlyValid, providerOutcomeUnknown: result.providerOutcomeUnknown, knownChargeMicros: reservation.chargedMicros, conservativeLiabilityMicros: result.liabilityMicros, officialDeltaUsd, noPlaintextKey: true, rawGamePass: false };
fs.writeFileSync(path.join(dir, outputFile), JSON.stringify(verification, null, 2), { flag: 'wx' });
console.log(JSON.stringify(verification));
