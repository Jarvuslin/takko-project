param([Parameter(Mandatory=$true)][string]$OutputPath)
$ErrorActionPreference = 'Stop'
if (Test-Path -LiteralPath $OutputPath) { throw 'Preserve previous key snapshots; use a fresh path.' }
$taskSecure = ConvertTo-SecureString ([IO.File]::ReadAllText((Join-Path $env:LOCALAPPDATA 'Takko\secrets\openrouter-testing.dpapi')))
$taskPointer = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($taskSecure)
try {
  $taskSecret = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($taskPointer)
  try { $taskResponse = Invoke-RestMethod -Uri 'https://openrouter.ai/api/v1/key' -Headers @{ Authorization = ('Bearer ' + $taskSecret) } } catch { throw 'Key status request failed; credential suppressed.' }
  $taskSafe = [ordered]@{ checkedAt = [DateTime]::UtcNow.ToString('o'); limit = $taskResponse.data.limit; usage = $taskResponse.data.usage; remaining = $taskResponse.data.limit_remaining; expiresAt = $taskResponse.data.expires_at }
  $taskJson = $taskSafe | ConvertTo-Json
  [IO.File]::WriteAllText([IO.Path]::GetFullPath($OutputPath), $taskJson)
  $taskJson
} finally { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($taskPointer); $taskSecret = $null; $taskSecure.Dispose() }
