import fs from "node:fs";
import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { parseJson } from "../src/generation/providers";
import { componentAdaptationDecisionSchema, validateComponentAdaptation } from "../src/generation/component-adaptation";
import { componentStageContext } from "../src/generation/asset-pipeline";
const root = "research/results/component-stage-v1";
const read = (f: string) => JSON.parse(fs.readFileSync(root + "/" + f, "utf8"));
const hash = (v: string | Buffer) => createHash("sha256").update(v).digest("hex");
const protocol = read("live/protocol.json"), result = read("live/result.json"), calls = read("live/calls.json");
const input = read("live/input.json"), cutoff = read("live/cutoff.json"), run = read("live/pre-adaptation-run.json");
const original = JSON.parse(fs.readFileSync("benchmarks/runs/marketplace-diversity-v10-20260916/combat-training/final-project.json", "utf8"));
const planned = JSON.parse(fs.readFileSync("benchmarks/runs/marketplace-diversity-v10-20260916/combat-training/plan-project.json", "utf8"));
assert.deepEqual(original.spec, planned.spec); assert.deepEqual(original.assetPipeline.needs, planned.spec.assetNeeds);
const events = original.assetPipeline.events;
assert.equal(cutoff.index, events.findIndex((e: any) => e.step === "component_adaptation_decision_call"));
assert.equal(cutoff.eventPrefixSha256, hash(JSON.stringify(events.slice(0, cutoff.index))));
assert.deepEqual(run.events, events.slice(0, cutoff.index));
assert(run.entries.every((e: any) => e.status === "pending" && !e.reason && !e.component && !e.selected));
assert.equal(run.entries.find((e: any) => e.needId === input.context.need.id).attempts, 1);
assert(run.entries.filter((e: any) => e.needId !== input.context.need.id).every((e: any) => e.attempts === 0));
assert.deepEqual(input.context.review, events.find((e: any) => e.step === "component_review_result").data);
assert.deepEqual(input.context.stage, componentStageContext(run, input.context.need, input.context.stage.current.candidate));
assert.equal(input.context.stage.current.candidate.id, input.context.evidence.candidateId);
assert.equal(events[cutoff.index].needId, input.context.need.id);
assert.equal(input.context.stage.namespace, original.scope);
assert.deepEqual(input.context.stage.needs.map((n: any) => n.need.id), original.assetPipeline.needs.map((n: any) => n.id));
assert.deepEqual(input, read("offline/input.json"));
for (const [file, expected] of Object.entries(protocol.sources)) assert.equal(hash(fs.readFileSync(file)), expected, "Production source changed: " + file);
assert(!result.unknownLiability && result.noFurtherCallsPending && result.nativeOperations === 0 && result.rawGamePass === false);
assert.equal(result.conservativePriorMicros, protocol.priorMicros + result.chargedOrUnknownMicros);
let first: any, spend = 0;
const assessments: any[] = [];
for (const arm of ["gemini", "sol"]) {
  const p = read(`live/${arm}/project.json`), outcome = read(`live/${arm}/result.json`);
  assert.equal(p.jobId, null); assert.equal(p.reservedMicros, 0);
  assert.deepEqual(p.artifact, { files: [], scene: [], coverage: [], assets: [] });
  const wire = read(`live/${arm}/call-1-request.json`);
  assert.deepEqual(wire, read(`offline/${arm}/call-1-request.json`));
  if (first) { assert.deepEqual(wire.messages, first.messages); assert.deepEqual(wire.response_format, first.response_format); } else first = wire;
  const rows = calls.filter((c: any) => c.arm === arm);
  assert.equal(rows.length, outcome.calls); assert.equal(p.charges.length, rows.length);
  assert(p.charges.every((c: any) => c.billingSource === "provider" && c.phase === "builder"));
  assert.equal(p.charges.reduce((s: number, c: any) => s + c.chargedMicros, 0), outcome.chargedOrUnknownMicros);
  for (const call of rows) {
    const response = read(`live/${arm}/call-${call.id}-response.txt`);
    assert.equal(call.chargedMicros, Math.ceil(response.usage.cost * 1e6));
    assert(spend + call.reserveMicros <= protocol.capMicros && protocol.priorMicros + spend + call.reserveMicros + protocol.headroomMicros <= protocol.campaignCeilingMicros);
    spend += call.chargedMicros;
    let decision: any, error: string | undefined;
    try { decision = componentAdaptationDecisionSchema.parse(parseJson(response.choices[0].message.content)); if (decision.action === "adapt") validateComponentAdaptation(decision.plan, input.context.evidence); } catch(e) { error = (e as Error).message; }
    if (call.id === rows.length) { assert.equal(!error, outcome.contractPassed); if (!error) assert.deepEqual(decision, read(`live/${arm}/decision.json`)); }
    assessments.push({ arm, attempt: call.id, responseHash: hash(response.choices[0].message.content), error: error ?? null });
  }
}
assert.equal(spend, result.chargedOrUnknownMicros);
const afterName = process.argv[2] ?? "key-after.json";
const before = read("live/key-before.json"), after = read("live/" + afterName);
const costUSD = calls.reduce((s: number, c: any) => s + c.costUSD, 0);
assert(Math.abs(after.usage - before.usage - costUSD) < 0.000002, "Provider aggregate has not reconciled");
for (const file of fs.readdirSync(root + "/live", { recursive: true, withFileTypes: true })) {
  if (!file.isFile()) continue;
  const text = fs.readFileSync(file.parentPath + "/" + file.name).toString("utf8");
  assert(!/sk-or-v1-[a-f0-9]{30,}/i.test(text), "Plaintext credential found; suppressed");
}
const verification = { at: new Date().toISOString(), passed: true, initialModelInputsEqual: true, originalReviewOnly: true, chronologyVerified: true, rawDecisionsChanged: false, assessments, chargedMicros: spend, costUSD, remaining: after.remaining, nativeOperations: 0, rawGamePass: false };
fs.writeFileSync(root + "/verification.json", JSON.stringify(verification, null, 2) + "\n", { flag: "wx" });
console.log(JSON.stringify(verification));
